SELECT
    lr.request_id,
    e.employee_code,
    e.full_name AS employee_name,
    e.email,
    e.department,
    e.designation,
    lt.leave_type_name,
    lr.start_date,
    lr.end_date,
    lr.requested_days,
    lr.reason,
    lr.status,
    lr.submitted_at,
    lr.decided_at
FROM public.leave_requests lr
JOIN public.employees e
    ON e.employee_id = lr.employee_id
JOIN public.leave_types lt
    ON lt.leave_type_id = lr.leave_type_id
WHERE lr.request_id =
    {{Number(tblHRLeaveRequests.selectedRow.request_id)}}::BIGINT;
