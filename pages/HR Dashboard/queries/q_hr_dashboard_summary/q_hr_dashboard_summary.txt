SELECT
    (
        SELECT COUNT(*)
        FROM public.employees
        WHERE active = TRUE
    ) AS total_employees,

    (
        SELECT COUNT(*)
        FROM public.leave_requests
        WHERE status = 'Pending'
    ) AS pending_requests,

    (
        SELECT COUNT(*)
        FROM public.leave_requests
        WHERE status = 'Approved'
    ) AS approved_requests,

    (
        SELECT COUNT(*)
        FROM public.leave_requests
        WHERE status = 'Rejected'
    ) AS rejected_requests;
