SELECT
    lr.request_id,
    e.employee_code,
    e.full_name AS employee_name,
    lt.leave_type_name,
    lr.start_date,
    lr.end_date,
    lr.requested_days,
    lr.reason,
    lr.status,
    lr.submitted_at,
    lr.decided_at
FROM public.leave_requests lr
JOIN public.employees e
    ON e.employee_id = lr.employee_id
JOIN public.leave_types lt
    ON lt.leave_type_id = lr.leave_type_id
WHERE
    (
        {{selHRRequestStatus.selectedOptionValue}} = 'All'
        OR lr.status = {{selHRRequestStatus.selectedOptionValue}}
    )
    AND
    (
        NULLIF(TRIM({{inpHRSearchRequest.text}}), '') IS NULL
        OR CAST(lr.request_id AS TEXT)
            ILIKE '%' || {{inpHRSearchRequest.text}} || '%'
        OR e.employee_code
            ILIKE '%' || {{inpHRSearchRequest.text}} || '%'
        OR e.full_name
            ILIKE '%' || {{inpHRSearchRequest.text}} || '%'
    )
ORDER BY lr.submitted_at DESC;
